A. Add a global auth gate middleware (UI + API)

Create server/middleware/authGate.ts:

// server/middleware/authGate.ts
import type { Express, Request, Response, NextFunction } from "express";

const PUBLIC_HTML_PATHS = new Set<string>([
  "/", "/login", "/2fa", "/2fa-fallback", "/forgot", "/reset",
]);

// Anything under /api/auth stays public; everything else under /api requires auth
const API_PUBLIC_PREFIXES = [/^\/api\/auth(\/|$)/, /^\/healthz$/, /^\/readyz$/, /^\/metrics$/];

function isHtmlRequest(req: Request) {
  const accept = (req.headers["accept"] || "").toString();
  return accept.includes("text/html");
}

function isApiPublic(req: Request) {
  const path = req.path;
  return API_PUBLIC_PREFIXES.some((re) => re.test(path));
}

export function registerAuthGate(app: Express) {
  // 1) Gate SPA page navigation
  app.use((req: Request, res: Response, next: NextFunction) => {
    if (req.method !== "GET" || !isHtmlRequest(req)) return next();

    // Allow public landing/login screens
    if (PUBLIC_HTML_PATHS.has(req.path)) return next();

    // Block everything else unless logged in
    const loggedIn = !!(req.session as any)?.user;
    if (!loggedIn) return res.redirect(302, "/");
    next();
  });

  // 2) Gate APIs (except /api/auth/* and observability)
  app.use((req: Request, res: Response, next: NextFunction) => {
    if (!req.path.startsWith("/api")) return next();
    if (isApiPublic(req)) return next();

    const user = (req.session as any)?.user;
    if (!user) return res.status(401).json({ error: "Unauthorized" });
    next();
  });
}


Wire it in server/index.ts (right after sessions are set up and before routes):

// server/index.ts (add these near the top with other imports)
import { registerAuthGate } from "./middleware/authGate";

// ... inside the async bootstrap, after setupSession(app) and BEFORE registerRoutes/app route registration:
registerAuthGate(app);


This enforces the “hard stop”: all HTML routes except /, /login, /2fa*, /forgot, /reset redirect to / if not logged in. And all /api/* except /api/auth/* (and /healthz etc.) require a session.

2) Admin-only account creation (no public registration)

There is no /api/auth/register route (good). Let’s add a clean admin create user endpoint.

Patch server/admin-routes.ts – add a POST /api/admin/users:

// server/admin-routes.ts
import type { Express, Request, Response } from "express";
import { ensureAdmin } from "./middleware/ensureAdmin";
import { db } from "./db";
import { users } from "@shared/schema";
import { eq } from "drizzle-orm";
import crypto from "crypto";
import { hashPassword } from "./security/password"; // <-- add this

export function registerAdminRoutes(app: Express) {
  app.get("/api/admin/users", ensureAdmin, async (_req: Request, res: Response) => {
    const rows = await db.select().from(users);
    const safe = rows.map(r => ({ id: r.id, email: r.email, role: r.role as "user" | "admin", isActive: r.isActive, lastLoginAt: r.lastLoginAt, createdAt: r.createdAt }));
    res.json({ users: safe });
  });

  // NEW: create user
  app.post("/api/admin/users", ensureAdmin, async (req: Request, res: Response) => {
    const { email, password, role = "user" } = req.body || {};
    if (!email || !password) return res.status(400).json({ error: "email and password are required" });
    const existing = await db.query.users.findFirst({ where: eq(users.email, email) as any });
    if (existing) return res.status(409).json({ error: "User already exists" });

    const passwordHash = await hashPassword(password);
    const [row] = await db.insert(users).values({
      email,
      passwordHash,
      role,
      isActive: 1,
    }).returning();

    res.status(201).json({ user: { id: row.id, email: row.email, role: row.role } });
  });

  app.patch("/api/admin/users/:id", ensureAdmin, async (req: Request, res: Response) => {
    const { id } = req.params;
    const partial = req.body || {};
    const updates: any = {};
    if (typeof partial.role === "string") updates.role = partial.role;
    if (typeof partial.isActive === "number") updates.isActive = partial.isActive;
    if (partial.password) {
      updates.passwordHash = await hashPassword(partial.password);
    }

    if (!Object.keys(updates).length) return res.status(400).json({ error: "No updates" });
    await db.update(users).set(updates).where(eq(users.id, id));
    res.json({ ok: true });
  });

  app.post("/api/admin/users/:id/reset", ensureAdmin, async (_req: Request, res: Response) => {
    const token = crypto.randomBytes(24).toString("hex");
    const payload: any = { ok: true };
    if (process.env.ALLOW_TEST_OTP_RESPONSE === "true") payload.token = token;
    res.json(payload);
  });
}

3) Lead scraper that returns real law-firm data

You already have PhantomBuster integration (great for LinkedIn/etc.). I’ve added a Google Places provider you can enable instantly with a key, so the app can fetch real firms (name, phone, website, address) anywhere.

A. Add Google Places service

Create server/places-service.ts:

// server/places-service.ts
import type { CrmProspect } from "@shared/schema";

interface Place {
  place_id: string;
  name: string;
  formatted_address?: string;
}

interface PlaceDetails {
  website?: string;
  international_phone_number?: string;
  formatted_phone_number?: string;
}

export class GooglePlacesService {
  private readonly apiKey = process.env.GOOGLE_MAPS_API_KEY;
  private readonly base = "https://maps.googleapis.com/maps/api/place";

  private requireKey() {
    if (!this.apiKey) throw new Error("Missing GOOGLE_MAPS_API_KEY");
  }

  async discoverLawFirms(query: string, location?: string, max = 20): Promise<CrmProspect[]> {
    this.requireKey();
    const textQuery = encodeURIComponent(`${query} ${location || ""}`.trim());
    const url = `${this.base}/textsearch/json?query=${textQuery}&type=lawyer&key=${this.apiKey}`;

    const r = await fetch(url);
    if (!r.ok) throw new Error(`Places textsearch failed: ${r.status}`);
    const data = await r.json();

    const places: Place[] = (data.results || []).slice(0, max).map((p: any) => ({
      place_id: p.place_id,
      name: p.name,
      formatted_address: p.formatted_address,
    }));

    const prospects: CrmProspect[] = [];
    for (const p of places) {
      const details = await this.fetchDetails(p.place_id);
      const phone = details.international_phone_number || details.formatted_phone_number || undefined;
      prospects.push({
        // Normalized prospect for your CRM table
        id: `places-${p.place_id}`,
        fullName: p.name,
        email: undefined, // rarely provided by Places; keep undefined
        company: p.name,
        phone,
        website: details.website,
        linkedinUrl: null,
        source: "google-places",
        tags: JSON.stringify(["law-firm"]),
        status: "new",
        notes: p.formatted_address || null,
        lastContactedAt: null,
        createdAt: new Date(),
      } as unknown as CrmProspect);
    }

    return prospects;
  }

  private async fetchDetails(placeId: string): Promise<PlaceDetails> {
    this.requireKey();
    const url = `${this.base}/details/json?place_id=${encodeURIComponent(placeId)}&fields=website,formatted_phone_number,international_phone_number&key=${this.apiKey}`;
    const r = await fetch(url);
    if (!r.ok) throw new Error(`Places details failed: ${r.status}`);
    const data = await r.json();
    return (data.result || {}) as PlaceDetails;
  }
}

export const googlePlacesService = new GooglePlacesService();

B. Add an endpoint to use it

Patch server/crm-routes.ts:

// server/crm-routes.ts
import type { Express } from "express";
import { storage } from "./storage";
import { 
  createProspectSchema,
  updateProspectSchema,
  sendOutreachSchema,
  scheduleMeetingSchema,
  syncSheetsSchema,
  type InsertCrmProspect 
} from "@shared/schema";
import { phantomBusterService } from "./phantombuster-service";
import { calendlyService } from "./calendly-service";
import { googleSheetsService } from "./sheets-service";
import { googlePlacesService } from "./places-service"; // <-- add this
import { ensureAuthed } from "./middleware/ensureAdmin"; // <-- reuse ensureAuthed

export function registerCrmRoutes(app: Express) {
  // Gate the whole CRM API
  app.use("/api/crm", ensureAuthed); // <-- add this line

  // NEW: Google Places discovery
  app.post("/api/crm/discover/places", async (req, res) => {
    try {
      const { query = "law firms", location = "Florida", limit = 20 } = req.body || {};
      const prospects = await googlePlacesService.discoverLawFirms(query, location, Math.min(50, Number(limit) || 20));

      // Deduplicate by email/website before saving
      const existing = await storage.getCrmProspects();
      const seenEmails = new Set(existing.map(p => p.email?.toLowerCase()).filter(Boolean));
      const seenWeb = new Set(existing.map(p => (p.website || "").toLowerCase()).filter(Boolean));

      let saved = 0;
      for (const p of prospects) {
        const lowerEmail = p.email?.toLowerCase();
        const lowerWeb = (p as any).website?.toLowerCase();
        if ((lowerEmail && seenEmails.has(lowerEmail)) || (lowerWeb && seenWeb.has(lowerWeb))) continue;
        await storage.createCrmProspect(p as InsertCrmProspect);
        if (lowerEmail) seenEmails.add(lowerEmail);
        if (lowerWeb) seenWeb.add(lowerWeb);
        saved++;
      }

      res.json({ ok: true, found: prospects.length, saved });
    } catch (e: any) {
      console.error("discover/places failed:", e);
      res.status(400).json({ error: e.message || "Failed to discover law firms" });
    }
  });

  // (existing routes unchanged) ...
}
