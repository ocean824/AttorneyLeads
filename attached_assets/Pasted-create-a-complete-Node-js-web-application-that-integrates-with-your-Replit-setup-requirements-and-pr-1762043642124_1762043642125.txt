# Florida Arrest Records CRM Connector

A Node.js web application for scraping Florida arrest records and exporting in PhantomBuster-compatible CSV format.

## Replit Setup

### 1) Required Replit Modules
- Enable Node.js 20
- (Optional) Enable PostgreSQL 16 (or use SQLite fallback)

### 2) Core Environment (Secrets)
Copy all variables from `.env.example` to Replit Secrets

### 3) Database Setup
**A. Quick Start (SQLite) - RECOMMENDED**
DB_DRIVER = sqlite
SQLITE_FILE = florida_arrests.db

text

**B. Production (Postgres)**
DB_DRIVER = pg
DATABASE_URL = postgres://user:pass@host:5432/dbname

text
Run in Shell: `npm run db:push`

### 4) Run & Build
- **Dev**: Click Run → runs `npm run dev` on port 5000
- **Prod**: `npm run start`

### 5) Web Interface
Access your Replit URL to use the web interface for scraping and exporting arrest records.

## Features

- Web-based scraping interface
- Real-time progress tracking
- PhantomBuster CSV export
- Multiple county support
- CRM integration ready
- Automated scheduling

## API Endpoints

- `GET /` - Web interface
- `POST /api/scrape` - Start scraping job
- `GET /api/scrape/status/:jobId` - Check job status
- `GET /api/export/csv` - Download CSV
- `GET /api/records` - List records
- `GET /api/health` - Health check

import express from 'express';
import cors from 'cors';
import helmet from 'helmet';
import rateLimit from 'express-rate-limit';
import session from 'express-session';
import compression from 'compression';
import { fileURLToPath } from 'url';
import { dirname, join } from 'path';
import dotenv from 'dotenv';

// Load environment variables
dotenv.config();

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const app = express();
const PORT = process.env.PORT || 5000;

// Security middleware
app.use(helmet({
  contentSecurityPolicy: {
    directives: {
      defaultSrc: ["'self'"],
      styleSrc: ["'self'", "'unsafe-inline'", "https://cdnjs.cloudflare.com"],
      scriptSrc: ["'self'", "'unsafe-inline'"],
      imgSrc: ["'self'", "data:", "https:"],
    },
  },
}));

// CORS configuration
const allowedOrigins = process.env.ALLOWED_ORIGINS?.split(',') || [];
app.use(cors({
  origin: function (origin, callback) {
    if (!origin || allowedOrigins.indexOf(origin) !== -1) {
      callback(null, true);
    } else {
      callback(new Error('Not allowed by CORS'));
    }
  },
  credentials: true
}));

// Rate limiting
const limiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: parseInt(process.env.RATE_LIMIT_MAX) || 100,
  message: 'Too many requests from this IP'
});
app.use(limiter);

// Session configuration
app.use(session({
  secret: process.env.SESSION_SECRET || 'fallback-secret-change-in-production',
  resave: false,
  saveUninitialized: false,
  cookie: { 
    secure: process.env.NODE_ENV === 'production',
    maxAge: 24 * 60 * 60 * 1000 // 24 hours
  }
}));

// Body parsing and compression
app.use(express.json({ limit: '10mb' }));
app.use(express.urlencoded({ extended: true }));
app.use(compression());

// Serve static files
app.use(express.static(join(__dirname, 'public')));

// Database setup
import { initializeDatabase } from './database/db.js';
await initializeDatabase();

    status: 'OK',

import sqlite3 from 'sqlite3';
import { promisify } from 'util';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

class Database {
  constructor() {
    const dbPath = join(__dirname, '..', process.env.SQLITE_FILE || 'florida_arrests.db');
    this.db = new sqlite3.Database(dbPath);
    
    // Promisify methods
    this.run = promisify(this.db.run.bind(this.db));
    this.get = promisify(this.db.get.bind(this.db));
    this.all = promisify(this.db.all.bind(this.db));
  }

  async initialize() {
    await this.run(`
      CREATE TABLE IF NOT EXISTS arrest_records (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        inmate_id TEXT,
        full_name TEXT,
        first_name TEXT,
        last_name TEXT,
        booking_date TEXT,
        release_date TEXT,
        charges TEXT,
        bond_amount TEXT,
        county TEXT,
        facility TEXT,
        age INTEGER,
        gender TEXT,
        race TEXT,
        scraped_date TEXT,
        source_url TEXT,
        raw_data TEXT,
        crm_contact_id TEXT,
        crm_sync_date TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        UNIQUE(inmate_id, county, booking_date)
      )
    `);

    await this.run(`
      CREATE TABLE IF NOT EXISTS scrape_jobs (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        job_id TEXT UNIQUE,
        counties TEXT,
        status TEXT DEFAULT 'pending',
        records_found INTEGER DEFAULT 0,
        started_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        completed_at DATETIME,
        error_message TEXT
      )
    `);

    await this.run(`
      CREATE TABLE IF NOT EXISTS crm_sync_log (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        record_count INTEGER,
        sync_date DATETIME DEFAULT CURRENT_TIMESTAMP,
        crm_type TEXT,
        status TEXT,
        error_message TEXT
      )
    `);

    console.log('Database initialized successfully');
  }

  async close() {
    return new Promise((resolve, reject) => {
      this.db.close((err) => {
        if (err) reject(err);
        else resolve();
      });
    });
  }
}

export const db = new Database();

export async function initializeDatabase() {
  await db.initialize();
}

import express from 'express';
import { db } from '../database/db.js';
import { spawn } from 'child_process';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';
import { v4 as uuidv4 } from 'uuid';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const router = express.Router();

// Active scraping jobs
const activeJobs = new Map();

router.post('/start', async (req, res) => {
  try {
    const { counties = process.env.DEFAULT_COUNTIES } = req.body;
    const jobId = uuidv4();

    // Start Python scraper as subprocess
    const scraperPath = join(__dirname, '..', 'scraper', 'app.py');
    const pythonProcess = spawn('python', [scraperPath, '--counties', counties, '--job-id', jobId]);

    const job = {
      id: jobId,
      counties: counties.split(','),
      status: 'running',
      startTime: new Date(),
      records: 0,
      process: pythonProcess
    };

    activeJobs.set(jobId, job);

    // Insert job record
    await db.run(
      'INSERT INTO scrape_jobs (job_id, counties, status) VALUES (?, ?, ?)',
      [jobId, counties, 'running']
    );

    // Handle scraper output
    pythonProcess.stdout.on('data', (data) => {
      const output = data.toString();
      console.log(`Scraper ${jobId}: ${output}`);
      
      // Parse progress updates
      const progressMatch = output.match(/Scraped (\d+) records from (\w+)/);
      if (progressMatch) {
        job.records += parseInt(progressMatch[1]);
      }
    });

    pythonProcess.stderr.on('data', (data) => {
      console.error(`Scraper ${jobId} error: ${data}`);
    });

    pythonProcess.on('close', async (code) => {
      if (code === 0) {
        job.status = 'completed';
        await db.run(
          'UPDATE scrape_jobs SET status = ?, completed_at = CURRENT_TIMESTAMP, records_found = ? WHERE job_id = ?',
          ['completed', job.records, jobId]
        );
      } else {
        job.status = 'failed';
        await db.run(
          'UPDATE scrape_jobs SET status = ?, completed_at = CURRENT_TIMESTAMP, error_message = ? WHERE job_id = ?',
          ['failed', `Process exited with code ${code}`, jobId]
        );
      }
      activeJobs.delete(jobId);
    });

    res.json({
      jobId,
      status: 'started',
      message: `Scraping started for counties: ${counties}`,
      estimatedTime: '5-15 minutes'
    });

  } catch (error) {
    console.error('Error starting scrape job:', error);
    res.status(500).json({ error: 'Failed to start scraping job' });
  }
});

router.get('/status/:jobId', async (req, res) => {
  try {
    const { jobId } = req.params;
    const job = activeJobs.get(jobId);
    
    if (!job) {
      // Check database for completed/failed jobs
      const dbJob = await db.get(
        'SELECT * FROM scrape_jobs WHERE job_id = ?',
        [jobId]
      );
      
      if (dbJob) {
        return res.json({
          jobId,
          status: dbJob.status,
          recordsFound: dbJob.records_found,
          startedAt: dbJob.started_at,
          completedAt: dbJob.completed_at,
          error: dbJob.error_message
        });
      }
      
      return res.status(404).json({ error: 'Job not found' });
    }

    res.json({
      jobId,
      status: job.status,
      counties: job.counties,
      recordsFound: job.records,
      runningTime: Math.round((new Date() - job.startTime) / 1000),
      estimatedProgress: 'running'
    });

  } catch (error) {
    console.error('Error getting job status:', error);
    res.status(500).json({ error: 'Failed to get job status' });
  }
});

router.get('/jobs', async (req, res) => {
  try {
    const jobs = await db.all(`
      SELECT * FROM scrape_jobs 
      ORDER BY started_at DESC 
      LIMIT 10
    `);
    
    res.json(jobs);
  } catch (error) {
    console.error('Error fetching jobs:', error);
    res.status(500).json({ error: 'Failed to fetch jobs' });
  }
});

export default router;

import express from 'express';
import { db } from '../database/db.js';
import { createObjectCsvWriter } from 'csv-writer';
import { join } from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const router = express.Router();

router.get('/csv', async (req, res) => {
  try {
    const { county, limit = 10000 } = req.query;
    
    let query = 'SELECT * FROM arrest_records WHERE 1=1';
    const params = [];
    
    if (county && county !== 'all') {
      query += ' AND county = ?';
      params.push(county);
    }
    
    query += ' ORDER BY created_at DESC LIMIT ?';
    params.push(parseInt(limit));

    const records = await db.all(query, params);

    if (records.length === 0) {
      return res.status(404).json({ error: 'No records found for export' });
    }

    // PhantomBuster compatible CSV format
    const csvWriter = createObjectCsvWriter({
      path: 'temp-export.csv',
      header: [
        { id: 'inmate_id', title: 'inmate_id' },
        { id: 'full_name', title: 'full_name' },
        { id: 'first_name', title: 'first_name' },
        { id: 'last_name', title: 'last_name' },
        { id: 'booking_date', title: 'booking_date' },
        { id: 'release_date', title: 'release_date' },
        { id: 'charges', title: 'charges' },
        { id: 'bond_amount', title: 'bond_amount' },
        { id: 'county', title: 'county' },
        { id: 'facility', title: 'facility' },
        { id: 'age', title: 'age' },
        { id: 'gender', title: 'gender' },
        { id: 'race', title: 'race' },
        { id: 'scraped_date', title: 'scraped_date' },
        { id: 'source_url', title: 'source_url' },
        { id: 'raw_data', title: 'raw_data' }
      ]
    });

    await csvWriter.writeRecords(records);

    const timestamp = new Date().toISOString().split('T')[0];
    const filename = `florida-arrests-${timestamp}.csv`;

    res.setHeader('Content-Type', 'text/csv');
    res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);

    const fs = await import('fs');
    const fileStream = fs.createReadStream('temp-export.csv');
    fileStream.pipe(res);

    // Clean up temp file after send
    fileStream.on('end', () => {
      fs.unlinkSync('temp-export.csv');
    });

  } catch (error) {
    console.error('Export error:', error);
    res.status(500).json({ error: 'Failed to export CSV' });
  }
});

router.get('/stats', async (req, res) => {
  try {
    const stats = await db.all(`
      SELECT 
        county,
        COUNT(*) as record_count,
        MAX(created_at) as last_updated
      FROM arrest_records 
      GROUP BY county 
      ORDER BY record_count DESC
    `);

    const total = await db.get('SELECT COUNT(*) as total FROM arrest_records');
    
    res.json({
      totalRecords: total.total,
      byCounty: stats,
      lastExport: new Date().toISOString()
    });

  } catch (error) {
    console.error('Stats error:', error);
    res.status(500).json({ error: 'Failed to get statistics' });
  }
});

export default router;

import express from 'express';
import { db } from '../database/db.js';

const router = express.Router();

router.get('/', async (req, res) => {
  try {
    const { page = 1, limit = 50, county, search } = req.query;
    const offset = (page - 1) * limit;

    let query = `
      SELECT * FROM arrest_records 
      WHERE 1=1
    `;
    const params = [];

    if (county && county !== 'all') {
      query += ' AND county = ?';
      params.push(county);
    }

    if (search) {
      query += ' AND (full_name LIKE ? OR inmate_id LIKE ? OR charges LIKE ?)';
      const searchTerm = `%${search}%`;
      params.push(searchTerm, searchTerm, searchTerm);
    }

    query += ' ORDER BY created_at DESC LIMIT ? OFFSET ?';
    params.push(parseInt(limit), offset);

    const records = await db.all(query, params);
    const total = await db.get(`
      SELECT COUNT(*) as count FROM arrest_records 
      WHERE 1=1
      ${county && county !== 'all' ? ' AND county = ?' : ''}
      ${search ? ' AND (full_name LIKE ? OR inmate_id LIKE ? OR charges LIKE ?)' : ''}
    `, params.slice(0, -2));

    res.json({
      records,
      pagination: {
        page: parseInt(page),
        limit: parseInt(limit),
        total: total.count,
        totalPages: Math.ceil(total.count / limit)
      }
    });

  } catch (error) {
    console.error('Error fetching records:', error);
    res.status(500).json({ error: 'Failed to fetch records' });
  }
});

router.get('/counties', async (req, res) => {
  try {
    const counties = await db.all(`
      SELECT DISTINCT county, COUNT(*) as record_count
      FROM arrest_records 
      GROUP BY county 
      ORDER BY record_count DESC
    `);
    
    res.json(counties);
  } catch (error) {
    console.error('Error fetching counties:', error);
    res.status(500).json({ error: 'Failed to fetch counties' });
  }
});

export default router;
