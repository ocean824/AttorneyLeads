A) Observability Upgrade
1. New Observability Routes – server/observability.ts

These three endpoints are implementation-ready; you just need to wire registerObservabilityRoutes(app) into your Express server (e.g., in server/index.ts after your other routes).

// server/observability.ts
import type { Express, Request, Response } from "express";
import { sql } from "drizzle-orm";
import { db } from "./db";
import { arrestRecords, scrapeJobs } from "@shared/schema";
import { logger } from "./logger";

type HealthStatus = {
  status: "ok" | "error";
  checks?: Record<string, unknown>;
};

export function registerObservabilityRoutes(app: Express) {
  app.get("/healthz", async (_req: Request, res: Response) => {
    try {
      // Lightweight DB connectivity check
      await db.select({ id: arrestRecords.id }).from(arrestRecords).limit(1);
      const payload: HealthStatus = { status: "ok" };
      res.json(payload);
    } catch (error) {
      logger.error("healthz_failed", {
        errorMessage: (error as Error).message,
      });
      const payload: HealthStatus = { status: "error" };
      res.status(500).json(payload);
    }
  });

  app.get("/readyz", async (_req: Request, res: Response) => {
    try {
      const [recordsRow] = await db
        .select({ count: sql<number>`count(*)` })
        .from(arrestRecords);

      const [jobsRow] = await db
        .select({ count: sql<number>`count(*)` })
        .from(scrapeJobs);

      const hasOpenAI = !!process.env.AI_INTEGRATIONS_OPENAI_API_KEY;

      const payload: HealthStatus = {
        status: "ok",
        checks: {
          db: true,
          env: {
            openai: hasOpenAI,
          },
          recordCount: recordsRow?.count ?? 0,
          jobCount: jobsRow?.count ?? 0,
        },
      };

      res.json(payload);
    } catch (error) {
      logger.error("readyz_failed", {
        errorMessage: (error as Error).message,
      });
      const payload: HealthStatus = {
        status: "error",
        checks: {
          db: false,
        },
      };
      res.status(500).json(payload);
    }
  });

  app.get("/metrics", async (_req: Request, res: Response) => {
    res.setHeader("Content-Type", "text/plain; version=0.0.4");

    try {
      const [recordsRow] = await db
        .select({ count: sql<number>`count(*)` })
        .from(arrestRecords);
      const [jobsRow] = await db
        .select({ count: sql<number>`count(*)` })
        .from(scrapeJobs);

      const lines = [
        "# HELP attorneyleads_up Application up status.",
        "# TYPE attorneyleads_up gauge",
        "attorneyleads_up 1",
        "# HELP attorneyleads_records_total Total number of arrest records.",
        "# TYPE attorneyleads_records_total gauge",
        `attorneyleads_records_total ${recordsRow?.count ?? 0}`,
        "# HELP attorneyleads_scrape_jobs_total Total number of scrape jobs.",
        "# TYPE attorneyleads_scrape_jobs_total gauge",
        `attorneyleads_scrape_jobs_total ${jobsRow?.count ?? 0}`,
      ];

      res.send(lines.join("\n") + "\n");
    } catch (error) {
      logger.error("metrics_failed", {
        errorMessage: (error as Error).message,
      });

      const lines = [
        "# HELP attorneyleads_up Application up status.",
        "# TYPE attorneyleads_up gauge",
        "attorneyleads_up 0",
      ];

      res.status(500).send(lines.join("\n") + "\n");
    }
  });
}

server/logger.ts
// server/logger.ts
type LogLevel = "debug" | "info" | "warn" | "error";

interface LogContext {
  requestId?: string;
  route?: string;
  [key: string]: unknown;
}

const SENSITIVE_KEYS = [
  "password",
  "secret",
  "token",
  "apikey",
  "apiKey",
  "authorization",
];

function redact(value: unknown): unknown {
  if (typeof value === "string") {
    if (value.length > 8 && /[A-Za-z0-9]{16,}/.test(value)) {
      return "[redacted]";
    }
    return value;
  }

  if (Array.isArray(value)) {
    return value.map(redact);
  }

  if (value && typeof value === "object") {
    const obj = value as Record<string, unknown>;
    const clone: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(obj)) {
      if (SENSITIVE_KEYS.some((s) => k.toLowerCase().includes(s.toLowerCase()))) {
        clone[k] = "[redacted]";
      } else {
        clone[k] = redact(v);
      }
    }
    return clone;
  }

  return value;
}

function log(level: LogLevel, message: string, ctx?: LogContext) {
  const payload = {
    level,
    message,
    ts: new Date().toISOString(),
    ...(ctx ? redact(ctx) : {}),
  };
  // eslint-disable-next-line no-console
  console.log(JSON.stringify(payload));
}

export const logger = {
  debug: (message: string, ctx?: LogContext) => log("debug", message, ctx),
  info: (message: string, ctx?: LogContext) => log("info", message, ctx),
  warn: (message: string, ctx?: LogContext) => log("warn", message, ctx),
  error: (message: string, ctx?: LogContext) => log("error", message, ctx),
};

server/middleware/loggerMiddleware.ts
// server/middleware/loggerMiddleware.ts
import type { Request, Response, NextFunction } from "express";
import { randomUUID } from "crypto";
import { logger } from "../logger";

declare module "express-serve-static-core" {
  interface Request {
    requestId?: string;
  }
}

export function loggerMiddleware(req: Request, res: Response, next: NextFunction) {
  const requestId = (req.headers["x-request-id"] as string) || randomUUID();
  req.requestId = requestId;
  res.setHeader("X-Request-Id", requestId);

  const start = Date.now();
  res.on("finish", () => {
    const duration = Date.now() - start;
    logger.info("http_request", {
      requestId,
      method: req.method,
      path: req.path,
      statusCode: res.statusCode,
      durationMs: duration,
    });
  });

  next();
}

import type { Request, Response, NextFunction } from "express";

export function securityHeaders(_req: Request, res: Response, next: NextFunction) {
  res.setHeader("X-Frame-Options", "DENY");
  res.setHeader("X-Content-Type-Options", "nosniff");
  res.setHeader("Referrer-Policy", "strict-origin-when-cross-origin");
  res.setHeader("Cross-Origin-Opener-Policy", "same-origin");
  res.setHeader("Cross-Origin-Embedder-Policy", "require-corp");
  res.setHeader(
    "Content-Security-Policy",
    [
      "default-src 'self'",
      "script-src 'self'",
      "style-src 'self' 'unsafe-inline'",
      "img-src 'self' data:",
      "connect-src 'self' https://api.openai.com https://*.replit.com",
      "frame-ancestors 'none'",
    ].join("; ")
  );
  next();
}

// config/rateLimits.ts
export type RateLimitConfig = {
  windowMs: number;
  max: number;
};

export type RouteRateLimitMap = {
  [routeKey: string]: RateLimitConfig;
};

export const routeRateLimits: RouteRateLimitMap = {
  "GET /api/stats": { windowMs: 15 * 60 * 1000, max: 100 },
  "GET /api/analytics": { windowMs: 15 * 60 * 1000, max: 60 },
  "GET /api/records": { windowMs: 15 * 60 * 1000, max: 120 },

  "POST /api/scrape/start": { windowMs: 15 * 60 * 1000, max: 10 },
  "POST /api/ai/enhance": { windowMs: 15 * 60 * 1000, max: 10 },
  "POST /api/crm/discover": { windowMs: 15 * 60 * 1000, max: 10 },
  "POST /api/crm/outreach/send": { windowMs: 15 * 60 * 1000, max: 10 },
  "POST /api/crm/sheets/sync-prospects": { windowMs: 15 * 60 * 1000, max: 10 },
  "POST /api/crm/sheets/sync-records": { windowMs: 15 * 60 * 1000, max: 10 },

  DEFAULT: { windowMs: 15 * 60 * 1000, max: 100 },
};

// server/middleware/rateLimiter.ts
import type { Request, Response, NextFunction } from "express";
import { routeRateLimits, type RateLimitConfig } from "../../config/rateLimits";

type BucketKey = string;

interface BucketState {
  count: number;
  windowStart: number;
}

const buckets = new Map<BucketKey, BucketState>();

function makeRouteKey(method: string, path: string): string {
  return `${method.toUpperCase()} ${path}`;
}

function getConfig(method: string, path: string): RateLimitConfig {
  const key = makeRouteKey(method, path);
  if (routeRateLimits[key]) return routeRateLimits[key];
  return routeRateLimits.DEFAULT;
}

export function rateLimiter(req: Request, res: Response, next: NextFunction) {
  const now = Date.now();
  const ip = (req.ip || (req.headers["x-forwarded-for"] as string) || "unknown")
    .toString()
    .split(",")[0]
    .trim();

  const config = getConfig(req.method, req.path);
  const bucketKey: BucketKey = `${ip}:${req.method}:${req.path}`;
  const existing = buckets.get(bucketKey);

  if (!existing) {
    buckets.set(bucketKey, { count: 1, windowStart: now });
    return next();
  }

  if (now - existing.windowStart > config.windowMs) {
    existing.count = 1;
    existing.windowStart = now;
    buckets.set(bucketKey, existing);
    return next();
  }

  if (existing.count >= config.max) {
    return res.status(429).json({
      code: "RATE_LIMITED",
      message: "Too many requests, please slow down.",
    });
  }

  existing.count += 1;
  buckets.set(bucketKey, existing);
  return next();
}

// scripts/smoke-test.ts
import http from "http";
import { db } from "../server/db";
import { arrestRecords } from "@shared/schema";

async function checkDb() {
  const records = await db.select().from(arrestRecords).limit(1);
  console.log("DB check ok, sample rows:", records.length);
}

function checkEndpoint(path: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const req = http.get(
      {
        host: "127.0.0.1",
        port: parseInt(process.env.PORT || "5000", 10),
        path,
      },
      (res) => {
        console.log(`Smoke: ${path} ->`, res.statusCode);
        if (res.statusCode && res.statusCode < 500) {
          resolve();
        } else {
          reject(new Error(`Endpoint ${path} returned ${res.statusCode}`));
        }
      }
    );

    req.on("error", reject);
  });
}

async function main() {
  await checkDb();
  await checkEndpoint("/healthz");
  await checkEndpoint("/readyz");
  console.log("Smoke test passed.");
}

main().catch((err) => {
  console.error("Smoke test failed:", err);
  process.exit(1);
});

# Architecture

## Overview

AttorneyLeads is a single Replit workspace application composed of:

- **client/** – React + Vite SPA for dashboard, records, duplicates, CRM, analytics.
