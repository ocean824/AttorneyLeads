🔧 Patch 1 — Expand server HTML guard (HARD STOP on all app pages)

File: server/middleware/protectDashboard.ts

Replace the file contents with:

import type { Express, Request, Response, NextFunction } from "express";

export function registerDashboardGuard(app: Express) {
  // Any HTML navigation into these prefixes requires an authenticated session
  const PROTECTED_PREFIXES = ["/dashboard", "/crm", "/records", "/jobs", "/duplicates", "/analytics", "/admin"];

  app.use((req: Request, res: Response, next: NextFunction) => {
    const isHtml = (req.headers["accept"] || "").toString().includes("text/html");
    if (req.method === "GET" && isHtml && PROTECTED_PREFIXES.some(p => req.path.startsWith(p))) {
      const loggedIn = !!(req.session as any)?.user;
      if (!loggedIn) {
        // Landing page at "/" contains the login panel
        return res.redirect(302, "/");
      }
    }
    next();
  });
}


This guarantees that going directly to /crm, /analytics, etc., never renders for non-authed users—even on a hard refresh.

🔧 Patch 3 (Optional) — Admin-only user creation endpoint

File: server/admin-routes.ts

Add the import:

import { hashPassword } from "./security/password";


Add this route inside export function registerAdminRoutes(app: Express) { ... }:

  // Create a user (admin-only)
  app.post("/api/admin/users", ensureAdmin, async (req: Request, res: Response) => {
    const { email, password, role = "user" } = req.body || {};
    if (!email || !password) return res.status(400).json({ error: "email and password are required" });

    const existing = await db.query.users.findFirst({ where: eq(users.email, email) as any });
    if (existing) return res.status(409).json({ error: "User already exists" });

    const passwordHash = await hashPassword(password);
    const [row] = await db.insert(users).values({
      email,
      passwordHash,
      role,
      isActive: 1,
    }).returning();

    res.status(201).json({ user: { id: row.id, email: row.email, role: row.role } });
  });


This lets the admin add users while keeping registration closed to the public.
