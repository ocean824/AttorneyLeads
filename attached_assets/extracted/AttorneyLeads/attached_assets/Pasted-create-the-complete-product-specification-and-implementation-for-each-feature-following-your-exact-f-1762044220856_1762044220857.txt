# Product Requirements Document: Florida Arrest Records CRM Connector

## Overview
A web application that scrapes Florida arrest records from public sources, processes the data with AI enhancement, and exports in PhantomBuster-compatible CSV format for CRM integration.

## Tech Stack
- **Backend**: Node.js 20, Express.js, SQLite/PostgreSQL
- **Frontend**: React 18, vanilla CSS
- **AI**: OpenAI GPT-4 for data enhancement and validation
- **Scraping**: Python microservice with BeautifulSoup4
- **Database**: Drizzle ORM with schema migrations
- **Deployment**: Replit.com with environment secrets

## User Stories
1. As a legal professional, I want to scrape arrest records from multiple Florida counties
2. As a CRM user, I want export data in PhantomBuster-compatible CSV format
3. As an admin, I want AI-enhanced data validation and enrichment
4. As a user, I want real-time progress tracking for scraping jobs
5. As a system, I want automated scheduling of record updates

## Core Features
- Multi-county parallel scraping
- AI-powered data normalization
- Real-time job monitoring
- PhantomBuster CSV export
- CRM webhook integrations
- Automated record deduplication

## AI Enhancements
- Charge classification and categorization
- Name standardization and validation
- Data quality scoring
- Duplicate record detection
- Geographic data enrichment

-- Enable UUID extension for PostgreSQL
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Arrest records table
CREATE TABLE IF NOT EXISTS arrest_records (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    inmate_id VARCHAR(100),
    full_name VARCHAR(255) NOT NULL,
    first_name VARCHAR(100),
    last_name VARCHAR(100),
    booking_date DATE,
    release_date DATE,
    charges TEXT,
    bond_amount VARCHAR(50),
    county VARCHAR(100) NOT NULL,
    facility VARCHAR(200),
    age INTEGER,
    gender VARCHAR(20),
    race VARCHAR(50),
    source_url TEXT,
    raw_data TEXT,
    
    -- AI enhancement fields
    ai_enhanced BOOLEAN DEFAULT FALSE,
    confidence_score DECIMAL(3,2),
    charge_categories JSONB,
    normalized_charges TEXT,
    risk_level VARCHAR(20),
    
    -- CRM integration
    crm_contact_id VARCHAR(100),
    crm_sync_status VARCHAR(50) DEFAULT 'pending',
    
    -- Timestamps
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    -- Constraints
    UNIQUE(inmate_id, county, booking_date)
);

-- Scrape jobs table
CREATE TABLE IF NOT EXISTS scrape_jobs (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    job_id VARCHAR(100) UNIQUE NOT NULL,
    counties JSONB NOT NULL,
    status VARCHAR(50) DEFAULT 'pending',
    records_found INTEGER DEFAULT 0,
    ai_enhanced BOOLEAN DEFAULT TRUE,
    
    -- AI usage tracking
    openai_requests INTEGER DEFAULT 0,
    openai_tokens_used INTEGER DEFAULT 0,
    
    started_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP WITH TIME ZONE,
    error_message TEXT
);

-- CRM sync log
CREATE TABLE IF NOT EXISTS crm_sync_log (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    record_count INTEGER,
    crm_type VARCHAR(50),
    sync_type VARCHAR(50),
    status VARCHAR(50),
    error_message TEXT,
    sync_date TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- AI enhancement cache
CREATE TABLE IF NOT EXISTS ai_enhancement_cache (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    input_hash VARCHAR(64) UNIQUE NOT NULL,
    original_data JSONB NOT NULL,
    enhanced_data JSONB NOT NULL,
    model_used VARCHAR(50),
    tokens_used INTEGER,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Indexes for performance
CREATE INDEX IF NOT EXISTS idx_arrest_records_county_date ON arrest_records(county, booking_date);
CREATE INDEX IF NOT EXISTS idx_arrest_records_ai_enhanced ON arrest_records(ai_enhanced);
CREATE INDEX IF NOT EXISTS idx_arrest_records_crm_sync ON arrest_records(crm_sync_status);
CREATE INDEX IF NOT EXISTS idx_scrape_jobs_status ON scrape_jobs(status);
CREATE INDEX IF NOT EXISTS idx_ai_cache_hash ON ai_enhancement_cache(input_hash);

-- Update timestamp trigger
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_arrest_records_updated_at 
    BEFORE UPDATE ON arrest_records 
    FOR EACH ROW 
    EXECUTE FUNCTION update_updated_at_column();

import express from 'express';
import cors from 'cors';
import helmet from 'helmet';
import rateLimit from 'express-rate-limit';
import { drizzle } from 'drizzle-orm/node-postgres';
import { eq, desc, sql } from 'drizzle-orm';
import { v4 as uuidv4 } from 'uuid';
import OpenAI from 'openai';
import { spawn } from 'child_process';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const app = express();
const PORT = process.env.PORT || 5000;

// Initialize OpenAI
const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY
});

// In-memory storage for demo (replace with database in production)
const scrapeJobs = new Map();
const arrestRecords = new Map();

// Middleware
app.use(helmet());
app.use(cors());
app.use(express.json());

// Rate limiting
const limiter = rateLimit({

  max: 100

app.post('/api/scrape/start', async (req, res) => {
  try {
    const { counties, enableAI = true } = req.body;
    const jobId = uuidv4();

    const job = {
      jobId,
      counties,
      status: 'running',
      recordsFound: 0,
      aiEnhanced: enableAI,
      startTime: new Date()
    };

    scrapeJobs.set(jobId, job);

    // Start Python scraper
    const scraperPath = join(__dirname, '../scraper/app.py');
    const pythonProcess = spawn('python', [
      scraperPath, 
      '--counties', counties.join(','),
      '--job-id', jobId,
      '--ai', enableAI.toString()
    ]);

    pythonProcess.stdout.on('data', (data) => {
      const output = data.toString();
      console.log(`Scraper ${jobId}:`, output);
      
      // Parse progress updates
      const progressMatch = output.match(/Scraped (\d+) records/);
      if (progressMatch) {
        job.recordsFound = parseInt(progressMatch[1]);
      }
    });

    pythonProcess.on('close', async (code) => {
      if (code === 0) {
        job.status = 'completed';
        
        // AI enhancement if enabled
        if (enableAI) {
          await enhanceRecordsWithAI(jobId);
        }
      } else {
        job.status = 'failed';
      }
      job.completedAt = new Date();
    });

    res.json(job);
  } catch (error) {
    console.error('Error starting scrape:', error);
    res.status(500).json({ error: 'Failed to start scraping' });
  }
});

app.get('/api/scrape/status/:jobId', (req, res) => {
  const job = scrapeJobs.get(req.params.jobId);
  if (!job) {
    return res.status(404).json({ error: 'Job not found' });
  }
  res.json(job);
});

app.get('/api/records', (req, res) => {
  const { county, page = 1, limit = 50 } = req.query;
  let records = Array.from(arrestRecords.values());
  
  if (county && county !== 'all') {
    records = records.filter(record => record.county === county);
  }
  
  const start = (page - 1) * limit;
  const end = start + parseInt(limit);
  const paginatedRecords = records.slice(start, end);
  
  res.json({
    records: paginatedRecords,
    pagination: {
      page: parseInt(page),
      limit: parseInt(limit),
      total: records.length,
      totalPages: Math.ceil(records.length / limit)
    }
  });
});

app.get('/api/export/csv', (req, res) => {
  const { county, aiEnhanced = true } = req.query;
  let records = Array.from(arrestRecords.values());
  
  if (county && county !== 'all') {
    records = records.filter(record => record.county === county);
  }
  
  if (aiEnhanced) {
    records = records.filter(record => record.ai_enhanced);
  }
  
  // Generate PhantomBuster CSV
  const csvHeaders = [
    'inmate_id', 'full_name', 'first_name', 'last_name', 
    'booking_date', 'charges', 'bond_amount', 'county',
    'facility', 'age', 'gender', 'race', 'confidence_score',
    'charge_categories', 'risk_level'
  ].join(',');
  
  const csvRows = records.map(record => 
    csvHeaders.split(',').map(field => 
      `"${(record[field] || '').toString().replace(/"/g, '""')}"`
    ).join(',')
  );
  
  const csvContent = [csvHeaders, ...csvRows].join('\n');
  
  res.setHeader('Content-Type', 'text/csv');
  res.setHeader('Content-Disposition', 'attachment; filename=arrest_records.csv');
  res.send(csvContent);
});

app.post('/api/ai/enhance', async (req, res) => {
  try {
    const { records } = req.body;
    const enhancedRecords = await enhanceRecordsWithAI('batch', records);
    res.json(enhancedRecords);
  } catch (error) {
    console.error('AI enhancement error:', error);
    res.status(500).json({ error: 'AI enhancement failed' });
  }
});

// AI Enhancement Function
async function enhanceRecordsWithAI(jobId, records = null) {
  const recordsToEnhance = records || Array.from(arrestRecords.values())
    .filter(record => !record.ai_enhanced)
    .slice(0, 100); // Limit batch size

  const enhancedRecords = [];
  
  for (const record of recordsToEnhance) {
    try {
      const completion = await openai.chat.completions.create({
        model: process.env.AI_MODEL || "gpt-4o-mini",
        messages: [
          {
            role: "system",
            content: `You are a legal data analyst. Enhance arrest records with:
            - Standardize name format (LAST, FIRST MIDDLE)
            - Categorize charges (felony/misdemeanor, violent/property/drug)
            - Validate and format dates
            - Extract bond amount
            - Calculate confidence score (0-1)
            - Assess risk level (low/medium/high)
            Return JSON with enhanced fields.`
          },
          {
            role: "user",
            content: JSON.stringify(record)
          }
        ],
        response_format: { type: "json_object" },
        max_tokens: 2000
      });

      const enhancedData = JSON.parse(completion.choices[0].message.content);
      const enhancedRecord = {
        ...record,
        ...enhancedData,
        ai_enhanced: true,
        enhanced_at: new Date().toISOString()
      };

      arrestRecords.set(record.inmate_id, enhancedRecord);
      enhancedRecords.push(enhancedRecord);
      
    } catch (error) {
      console.error('AI enhancement failed for record:', record.inmate_id, error);
    }
  }

  return enhancedRecords;
}

                </div>
                
                <div className="job-status">
                  <p><strong>Status:</strong> {jobStatus.status}</p>
                  <p><strong>Records Found:</strong> {jobStatus.recordsFound}</p>
                  <p><strong>Counties:</strong> {jobStatus.counties?.join(', ')}</p>
                  <p><strong>AI Enhancement:</strong> {jobStatus.aiEnhanced ? 'Enabled' : 'Disabled'}</p>
                </div>

                {jobStatus.status === 'completed' && (
                  <div className="completion-actions">
                    <button onClick={downloadCSV} className="download-btn">
                      Download CSV for PhantomBuster
                    </button>
                    <button onClick={() => {
                      setActiveJob(null);
                      setJobStatus(null);
                      setCurrentStep(0);
                    }}>
                      Start New Scrape
                    </button>
                  </div>
                )}

                {jobStatus.status === 'failed' && (
                  <div className="error-message">
                    <p>Scraping failed. Please try again.</p>
                    <button onClick={() => {
                      setActiveJob(null);
                      setJobStatus(null);
                      setCurrentStep(0);
                    }}>
                      Retry
                    </button>
                  </div>
                )}
              </>
            )}
          </div>
        )}
      </div>
    </div>
  );
};

export default ScrapingWizard;
